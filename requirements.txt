import streamlit as st

from rag import create_multi_agent_workflow

st.set_page_config(page_title="AtomBot", page_icon="🤖", layout="centered")

st.title("AtomBot")
st.caption("Multi-Agent LangGraph Assistant")


@st.cache_resource
def load_agent():
    return create_multi_agent_workflow()


agent = load_agent()

if "messages" not in st.session_state:
    st.session_state.messages = []

for message in st.session_state.messages:
    with st.chat_message(message["role"]):
        st.write(message["content"])

question = st.chat_input("Ask a question about the document or current information...")

if question:
    st.session_state.messages.append({"role": "user", "content": question})

    with st.chat_message("user"):
        st.write(question)

    history = ""
    for message in st.session_state.messages[:-1]:
        history += f'{message["role"].capitalize()}: {message["content"]}\n'

    with st.chat_message("assistant"):
        with st.spinner("Thinking..."):
            try:
                result = agent.invoke(
                    {
                        "question": question,
                        "history": history,
                        "local_context": "",
                        "web_context": "",
                        "answer": "",
                    }
                )
            except FileNotFoundError as exc:
                st.error(str(exc))
                st.stop()

        answer = result.get("answer", "I could not generate an answer.")
        st.write(answer)

        with st.expander("View Sources"):
            source_parts = []
            if result.get("local_context"):
                source_parts.append("Local document context:\n" + result["local_context"])
            if result.get("web_context"):
                source_parts.append("Web search context:\n" + result["web_context"])

            if not source_parts:
                st.write("No supporting sources were used.")
            else:
                st.write("\n\n".join(source_parts))

    st.session_state.messages.append({"role": "assistant", "content": answer})
